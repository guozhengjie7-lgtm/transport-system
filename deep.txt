// 显示错误提示
function showError(containerId, message) {
  const el = document.getElementById(containerId);
  if (el) {
    el.innerText = message;
    setTimeout(() => { el.innerText = ""; }, 3000);
  } else {

  const password = document.getElementById('password').value;

// 切换页面
function show(id) {
  const pages = ['dashboard', 'vehicle', 'dispatch', 'finance'];
  pages.forEach(page => {
    const el = document.getElementById(page);
    if (el) el.style.display = page === id ? 'block' : 'none';
  });
}

// 加载所有数据
async function loadAll() {
  await loadVehicle();
  await loadDispatch();
  await loadFinance();
}

// 车辆管理
async function loadVehicle() {
  try {
    const { data, error } = await supabase.from('vehicles').select('*');
    if (error) throw error;
    const tbody = document.getElementById('vehicleTable');
    tbody.innerHTML = data.map(v => `<tr><td>${escapeHtml(v.plate)}</td><td>${escapeHtml(v.type)}</td></tr>`).join('');
    // 填充派车页的车牌下拉框
    const plateSelect = document.getElementById('d_plate');
    plateSelect.innerHTML = data.map(v => `<option value="${escapeHtml(v.plate)}">${escapeHtml(v.plate)}</option>`).join('');
  } catch (err) {
    showError('vehicleError', '加载车辆失败：' + err.message);
  }
}

async function addVehicle() {
  if (role !== "admin") return showError('vehicleError', '无权限，仅管理员可添加');
  const plate = document.getElementById('v_plate').value.trim();
  const type = document.getElementById('v_type').value;
  if (!plate) return showError('vehicleError', '请输入车牌号');

  try {
    const { error } = await supabase.from('vehicles').insert([{ plate, type }]);
    if (error) throw error;
    document.getElementById('v_plate').value = '';
    await loadVehicle();
  } catch (err) {
    showError('vehicleError', '添加失败：' + err.message);
  }
}

// 派车管理
async function loadDispatch() {
  try {
    const { data, error } = await supabase.from('dispatch').select('*');
    if (error) throw error;
    const tbody = document.getElementById('dispatchTable');
    tbody.innerHTML = data.map(d => `<tr><td>${escapeHtml(d.date)}</td><td>${escapeHtml(d.plate)}</td><td>${d.ton}</td></tr>`).join('');
    document.getElementById('totalDispatch').innerText = data.length;
    document.getElementById('totalTon').innerText = data.reduce((sum, d) => sum + (d.ton || 0), 0);
  } catch (err) {
    showError('dispatchError', '加载派车记录失败：' + err.message);
  }
}

async function addDispatch() {
  if (role !== "admin") return showError('dispatchError', '无权限，仅管理员可添加');
  const date = document.getElementById('d_date').value.trim();
  const plate = document.getElementById('d_plate').value;
  const ton = parseFloat(document.getElementById('d_ton').value);
  if (!date || !plate || isNaN(ton)) return showError('dispatchError', '请完整填写日期、车牌和吨数');

  try {
    const { error } = await supabase.from('dispatch').insert([{ date, plate, ton }]);
    if (error) throw error;
    document.getElementById('d_date').value = '';
    document.getElementById('d_ton').value = '';
    await loadDispatch();
  } catch (err) {
    showError('dispatchError', '添加失败：' + err.message);
  }
}

// 资金管理
async function loadFinance() {
  try {
    const { data, error } = await supabase.from('finance').select('*');
    if (error) throw error;
    const tbody = document.getElementById('financeTable');
    tbody.innerHTML = data.map(f => `<tr><td>${escapeHtml(f.project)}</td><td>${f.total}</td><td>${f.paid}</td></tr>`).join('');
    const receivable = data.reduce((sum, f) => sum + (f.total || 0), 0);
    const received = data.reduce((sum, f) => sum + (f.paid || 0), 0);
    document.getElementById('receivable').innerText = receivable;
    document.getElementById('received').innerText = received;
  } catch (err) {
    showError('financeError', '加载资金数据失败：' + err.message);
  }
}

async function addFinance() {
  if (role !== "admin") return showError('financeError', '无权限，仅管理员可添加');
  const project = document.getElementById('f_project').value.trim();
  const total = parseFloat(document.getElementById('f_total').value);
  const paid = parseFloat(document.getElementById('f_paid').value);
  if (!project || isNaN(total) || isNaN(paid)) return showError('financeError', '请完整填写项目、总金额和已收金额');

  try {
    const { error } = await supabase.from('finance').insert([{ project, total, paid }]);
    if (error) throw error;
    document.getElementById('f_project').value = '';
    document.getElementById('f_total').value = '';
    document.getElementById('f_paid').value = '';
    await loadFinance();
  } catch (err) {
    showError('financeError', '添加失败：' + err.message);
  }
}

// 辅助函数：防止XSS
function escapeHtml(str) {
  if (!str) return '';
  return str.replace(/[&<>]/g, function(m) {
    if (m === '&') return '&amp;';
    if (m === '<') return '&lt;';
    if (m === '>') return '&gt;';
    return m;
  });
}
